511

605def park_pipeline_state(state: GrugMoeAutomaticPipelineState) -> ParkedPipelineState:
606    """Free device memory for disposable warmup by parking real state on host.
607 
608    Device buffers are invalidated after their host copies complete. Callers must
609    discard aliases of the original state and restore the returned state before
610    training. Already-host-resident optimizer arrays remain intact.
611    """
612    pp, _ = _jaxpp_modules()
613    originals, tree = jax.tree.flatten(state)
614    parked = []
615    original_shardings = []
616    device_arrays = {}
617    local_device_bytes = 0
618    for value in originals:
619        assert isinstance(value, pp.MpmdArray)
620        original = value._mpmd_sharding
621        original_shardings.append(original)
622        if original.memory_kind == _HOST_MEMORY_KIND:
623            parked.append(value)
624            continue
625        local = value.to_mpmd_local_array
626        assert local is None or isinstance(local, jax.Array), "Each process must own one pipeline stage"
627        host_arrays = []
628        if local is not None:
629            host = _copy_array_to_host(local)
630            host.block_until_ready()
631            host_arrays.append(host)
632            if id(local) not in device_arrays:
633                local_device_bytes += sum(shard.data.nbytes for shard in local.addressable_shards)
634                device_arrays[id(local)] = local
635        parked.append(
636            pp.MpmdArray(
637                host_arrays,
638                dataclasses.replace(original, memory_kind=_HOST_MEMORY_KIND),
639                shape=value.shape,
640                dtype=value.dtype,
641            )
642        )
643    # Finish every copy before deleting anything: aliases can occur across state leaves.
644    for array in device_arrays.values():
645        array.delete()
646    return ParkedPipelineState(jax.tree.unflatten(tree, parked), tuple(original_shardings), local_device_bytes)

119def _initialize_pipeline_communicators(mpmd_mesh, placements: tuple[int, ...]) -> None:
120    # Create every neighboring channel before a blocking NCCL group can prevent
121    # a stage from reaching another stage's communicator initialization.
122    edges = sorted({tuple(sorted((a, b))) for a, b in itertools.pairwise(placements) if a != b})
123    for left, right in edges:
124        for source, target in zip(
125            mpmd_mesh.unstack[left].devices.flat, mpmd_mesh.unstack[right].devices.flat, strict=True
126        ):
127            if jax.process_index() not in (source.process_index, target.process_index):
128                continue
129            if env_vars.jaxpp_directional_communicators.value:
130                dime2.get_or_create_comm(dime2.UniqueDevices(source, target))
131                dime2.get_or_create_comm(dime2.UniqueDevices(target, source))
132            else:
133                dime2.get_or_create_comm(dime2.UniqueSortedDevices(source, target))
134    barrier_sync_named("hero_pipeline_communicators_initialized", timeout=_MULTIHOST_TIMEOUT)
135 
136    # NCCL establishes P2P transports on the first send/recv, even after
137    # communicator initialization. Connect both directions in a common edge
138    # order before pipeline scheduling can introduce a blocking setup cycle.
139    for left, right in edges:
140        left_mesh, right_mesh = mpmd_mesh.unstack[left], mpmd_mesh.unstack[right]
141        if any(device.process_index == jax.process_index() for device in left_mesh.devices.flat):
142            local_stage, remote_stage = left, right
143        elif any(device.process_index == jax.process_index() for device in right_mesh.devices.flat):
144            local_stage, remote_stage = right, left
145        else:
146            continue
147        local_sharding = NamedSharding(mpmd_mesh.unstack[local_stage], P())
148        remote_sharding = NamedSharding(mpmd_mesh.unstack[remote_stage], P())
149        payload = jax.make_array_from_callback(
150            (1,), local_sharding, lambda _, stage=local_stage: np.full((1,), stage, dtype=np.float32)
151        )
152        recv_buffer = jax.make_array_from_callback((1,), local_sharding, lambda _: np.zeros((1,), dtype=np.float32))
153        transfer = dime2.start_transfer([payload], [remote_sharding], [recv_buffer], [remote_sharding])
154        (received,) = transfer.done()
155        received.block_until_ready()

157            actual = np.asarray(shard.data)
158            if not np.all(actual == remote_stage):
159                raise RuntimeError(f"Pipeline channel {remote_stage}->{local_stage} received {actual}")
160    barrier_sync_named("hero_pipeline_channels_connected", timeout=_MULTIHOST_TIMEOUT)

4"""Train Snowball on synthetic tokens through the Levanter trainer and report step throughput.
5 
6A single-process, one-node harness (for example one Slurm job) that measures Snowball's real training
7step: mixed precision, Adam, the model's per-layer remat, and the trainer's own throughput and MFU
8logging. It uses the portable code paths only: reference attention, and XLA ragged_dot when
9``RAGGED_DOT_IMPL=xla`` is set, which AMD GPUs need because the default kernels are CUDA-only.
10

13targets. Checkpoints are never written. The trainer's forced final save of the 67B model plus Adam state would
14be about 1 TB, so the default hooks are replaced with the logging and profiler hooks only.

17fails in the next-token loss (see ``main``).

>) 
210    if args.layers is not None:
211        model_cfg = dataclasses.replace(model_cfg, num_layers=args.layers)
212    seq_len = preset.seq_len if args.seq_len is None else args.seq_len
213    batch_size = jax.device_count() if args.batch_size is None else args.batch_size
214    if batch_size == 1:
215        # jnp.roll in the next-token loss slices the (1, seq_len) token grid to (1, 1), and under an explicit mesh JAX
216        # drops the sharding of that slice, then rejects concatenating it with the (1, seq_len - 1) remainder.
217        raise ValueError("a global batch of 1 fails in the loss under explicit mesh axes; pass --batch-size 2 or more")
218    run_id = args.run_id or f"snowball-{args.size}-{time.strftime('%Y%m%d-%H%M%S')}"
219 
220    trainer_cfg = build_trainer_config(args, preset, run_id, batch_size)
221    profile = args.profile_steps > 0
222 
223    def loss_function(model, example, *, key=None):
224        return model.compute_next_token_loss(example, key=key)
225 
226    levanter.trainer.initialize(trainer_cfg)
227    logger.info(
228        "size=%s layers=%d hidden=%d experts=%d topk=%d batch=%d seq_len=%d mp=%s expert_axis=%d context_axis=%d",
229        args.size,
230        model_cfg.num_layers,
231        model_cfg.hidden_dim,
232        model_cfg.num_experts,
233        model_cfg.num_experts_per_token,
234        batch_size,
235        seq_len,
236        trainer_cfg.mp,
237        args.expert_axis,
238        args.context_axis,
239    )

241    step_durations: list[float] = []

243    def record_step(info: StepInfo, force: bool = False) -> None:

245        if not force and info.step >= FIRST_TIMED_STEP:
246            step_durations.append(info.step_duration)
247 
248    with Trainer(trainer_cfg, optimizer, loss_function, add_default_hooks=False) as trainer:

250        Vocab = Axis("vocab", model_cfg.vocab_size)

258        )

260        if profile:

262            profile_dir = str(trainer_cfg.log_dir / trainer.run_id / "profiler")
