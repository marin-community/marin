65def mixture_nll(
66    native_nll: jax.Array,
67    log_partition: jax.Array,
68    copy_logit: jax.Array,
69    sink_logp: jax.Array,
70    target_logp: jax.Array,
71) -> jax.Array:
72    """Mix the vocabulary head, copy target mass and sink fallback in log space.
73 
74    The copy gate competes with the whole vocabulary: sigmoid(copy_logit-log Z).
75    All inputs and the result have shape (batch, sequence). Keep log Z attached
76    to autodiff; detaching it changes the vocabulary-head training objective.

14def dense_reference(params, tokens, labels, segments):
15    hidden, vocabulary, query, key, gate, sink = params
16    logits = hidden @ vocabulary

18    copy_gate = jax.nn.softmax(jnp.concatenate([logits, (hidden @ gate)[..., None]], -1), -1)[..., -1]
19    queries, keys = hidden @ query, hidden @ key
20    scores = jnp.einsum("bqd,bkd->bqk", queries, keys) / hidden.shape[-1] ** 0.5
21    length = tokens.shape[1]
22    allowed = (jnp.arange(length)[None, :] <= jnp.arange(length)[:, None])[None] & (
23        segments[:, :, None] == segments[:, None, :]
24    )
25    scores = jnp.where(allowed, scores, -jnp.inf)
26    sink_score = (queries @ sink) / hidden.shape[-1] ** 0.5
27    attention = jax.nn.softmax(jnp.concatenate([scores, sink_score[..., None]], -1), -1)

4"""Train Snowball on synthetic tokens through the Levanter trainer and report step throughput.
5 
6A single-process, one-node harness (for example one Slurm job) that measures Snowball's real training
7step: mixed precision, Adam, the model's per-layer remat, and the trainer's own throughput and MFU
8logging. By default it uses the portable code paths: reference attention, and XLA ragged_dot when
9``RAGGED_DOT_IMPL=xla`` is set, which AMD GPUs need because the default kernels are CUDA-only. ``--attention``
10selects any Grug attention implementation.
11 
12Every step sees fresh uniform-random tokens and no example repeats, so no model can get below a loss of
13ln(vocab_size) (11.76 at full vocab). A loss that drops below that floor means something is leaking the
14targets. Checkpoints are never written. The trainer's forced final save of the 67B model plus Adam state would
15be about 1 TB, so the default hooks are replaced with the logging and profiler hooks only.
16 
17The default batch is one sequence per device. On a single device pass ``--batch-size 2``: a global batch of 1
18fails in the next-token loss (see ``main``).
19 
20Example, 8 GPUs, full 67B shape, one 4096-token sequence per GPU, profile of steps 10-12::
21 
22    RAGGED_DOT_IMPL=xla python experiments/grug/snowball_synthetic/train.py \\
23        --size full --steps 20 --profile-steps 3
24"""

160def build_trainer_config(args: argparse.Namespace, preset: Preset, run_id: str, batch_size: int) -> TrainerConfig:
161    compute_mapping = {"batch": ["replica_dcn", "data", "expert"], "vocab": "model"}
162    if args.context_axis > 1:
163        # Snowball only shards activations over a context axis wider than one device; the loss must agree.
164        compute_mapping["position"] = "context"
165    mesh = MeshConfig(
166        axes={"data": -1, "replica": 1, "model": 1, "context": args.context_axis, "expert": args.expert_axis},
167        compute_mapping=compute_mapping,
168    )
169    return TrainerConfig(
170        id=run_id,
171        log_dir=args.log_dir,
172        mesh=mesh,
173        use_explicit_mesh_axes=True,
174        # The tiny preset is a CPU check; the default demands an accelerator everywhere except macOS.
175        require_accelerator=False,
176        mp=jmp.get_policy(args.mp or preset.mp),
177        train_batch_size=batch_size,
178        num_train_steps=args.steps,
179        tracker=JsonLoggerConfig(),
180        # Never consulted for saves (the checkpoint hook is not installed); only names the resume-search root.
181        checkpointer=CheckpointerConfig(base_path=str(args.log_dir / run_id / "checkpoints")),
182        distributed=DistributedConfig(initialize_jax_distributed=False),
183        profiler=ProfilerConfig(
184            enabled=args.profile_steps > 0,
185            start_step=args.profile_start,
186            num_steps=args.profile_steps,
187            upload=XprofUploadConfig(enabled=False),
188        ),
189        jax_compilation_cache_dir=args.compilation_cache_dir,
190        log_jaxprs=False,
191        log_xla_hlo=False,
192    )

195def log_throughput_summary(
196    step_durations: list[float], batch_size: int, seq_len: int, flops_per_example: float, num_steps: int
197) -> None:
198    median = statistics.median(step_durations)
199    throughput = compute_instant_throughput(batch_size, median, seq_len, flops_per_example, aggregate_device_flops())
200    logger.info(
201        "median step over steps %d-%d: %.3f s  %s tokens/s  %s model TFLOP/s  MFU %s%%",
202        FIRST_TIMED_STEP,
203        num_steps - 1,
204        median,
205        f"{throughput.tokens_per_second:,.0f}" if throughput.tokens_per_second is not None else "n/a",
206        f"{throughput.model_flops_per_second / 1e12:.1f}" if throughput.model_flops_per_second is not None else "n/a",
207        f"{throughput.mfu:.2f}" if throughput.mfu is not None else "n/a",
208    )

211def main() -> None:
212    args = parse_args()
213    preset = PRESETS[args.size]
214    model_cfg = dataclasses.replace(
215        preset.model, attention_implementation=args.attention, moe_implementation=args.moe_impl
216    )
217    if args.layers is not None:
218        model_cfg = dataclasses.replace(model_cfg, num_layers=args.layers)
219    seq_len = preset.seq_len if args.seq_len is None else args.seq_len
220    batch_size = jax.device_count() if args.batch_size is None else args.batch_size
221    if batch_size == 1:
222        # jnp.roll in the next-token loss slices the (1, seq_len) token grid to (1, 1), and under an explicit mesh JAX
223        # drops the sharding of that slice, then rejects concatenating it with the (1, seq_len - 1) remainder.
224        raise ValueError("a global batch of 1 fails in the loss under explicit mesh axes; pass --batch-size 2 or more")
225    run_id = args.run_id or f"snowball-{args.size}-{time.strftime('%Y%m%d-%H%M%S')}"
226 
227    trainer_cfg = build_trainer_config(args, preset, run_id, batch_size)
228    profile = args.profile_steps > 0
229 
230    def loss_function(model, example, *, key=None):
231        return model.compute_next_token_loss(example, key=key)
232 
233    levanter.trainer.initialize(trainer_cfg)
234    logger.info(
235        "size=%s layers=%d hidden=%d experts=%d topk=%d batch=%d seq_len=%d mp=%s expert_axis=%d context_axis=%d",
236        args.size,
237        model_cfg.num_layers,
238        model_cfg.hidden_dim,
239        model_cfg.num_experts,
240        model_cfg.num_experts_per_token,
241        batch_size,
242        seq_len,
243        trainer_cfg.mp,
244        args.expert_axis,
245        args.context_axis,
246    )
247    optimizer = AdamConfig(learning_rate=args.learning_rate, warmup=0.0).build(args.steps)
248    step_durations: list[float] = []
249 
250    def record_step(info: StepInfo, force: bool = False) -> None:
251        # The trainer re-runs every hook with force=True after the last step; that call would duplicate the sample.
252        if not force and info.step >= FIRST_TIMED_STEP:
253            step_durations.append(info.step_duration)
254 
255    with Trainer(trainer_cfg, optimizer, loss_function, add_default_hooks=False) as trainer:
256        Pos = model_cfg.max_Pos.resize(seq_len)
257        Vocab = Axis("vocab", model_cfg.vocab_size)
258        model_key, training_key = jrandom.split(jrandom.PRNGKey(trainer_cfg.seed))
259        flops_per_example = 3 * snowball_flops_per_token(model_cfg, Vocab.size, Pos.size) * Pos.size
260 
261        trainer.add_hook(callbacks.pbar_logger(total=args.steps), every=1)
262        trainer.add_hook(callbacks.log_step_info(args.steps, trainer_cfg.batch_schedule), every=1)
263        trainer.add_hook(
264            callbacks.log_performance_stats(Pos.size, trainer_cfg.batch_schedule, flops_per_example), every=1
265        )
266        trainer.add_hook(record_step, every=1)
267        if profile:
268            profile_steps = trainer_cfg.profiler.resolve_num_profile_steps(num_train_steps=args.steps)
269            profile_dir = str(trainer_cfg.log_dir / trainer.run_id / "profiler")
270            trainer.add_hook(
271                trainer_cfg.profiler.build(profile_dir, run_id=trainer.run_id, num_steps=profile_steps), every=1
272            )
273 
274        # Straight to the loader, not through LmDataConfig's mixer: the mixer draws in blocks of 2048 and
275        # wraps a shorter dataset back to its start, which would repeat examples and let the model memorize.
276        examples = synthetic_examples(args.steps * batch_size, seq_len, model_cfg.vocab_size)
277        train_dataset = NamedLmDataset(ListAsyncDataset(examples), Pos)
278        start = time.perf_counter()
279        state = trainer.initial_state(training_key, model_init=lambda: model_cfg.build(Vocab, key=model_key))
280        logger.info("initialized trainer state in %.1f s", time.perf_counter() - start)
281        report_memory()
282 
283        trainer.train(state, trainer.data_loader(train_dataset))
284        trainer.tracker.finish()
285 
286    if step_durations:
287        log_throughput_summary(step_durations, batch_size, seq_len, flops_per_example, args.steps)
288    report_memory()
289 
290 
291if __name__ == "__main__":
292    main()
