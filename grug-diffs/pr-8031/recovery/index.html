150# Model: a stack of dense hidden_dim x hidden_dim layers plus an "lm head".

155def init_params(key, *, num_layers: int, hidden_dim: int, vocab_size: int):

171def rms_norm(h, eps=1e-6):
172    """Paramless RMSNorm. The real model has RMSNorm + residual around every block;
173    without them a 48-layer stack at initializer_std=0.5/sqrt(d) decays to zero and
174    trains nothing (verified: loss pinned at exactly ln(vocab_size))."""
175    v = jnp.mean(jnp.square(h.astype(jnp.float32)), axis=-1, keepdims=True)
176    return (h.astype(jnp.float32) * jax.lax.rsqrt(v + eps)).astype(h.dtype)

179def forward(params, x, *, compute_dtype):
180    """Forward pass. Each `reshard(w, REPLICATED)` is one FSDP all-gather over `data`."""
181    h = x.astype(compute_dtype)
182    for w in params["layers"]:
183        # params=float32 -> compute=bfloat16 (launch.py MIXED_PRECISION),
184        # cast BEFORE the gather so the collective moves bf16, as jmp does.
185        w_full = reshard(w.astype(compute_dtype), REPLICATED)
186        h = h + jax.nn.gelu(rms_norm(h) @ w_full)  # pre-norm residual block
187        h = reshard(h, BATCH_SPEC)  # pin activations to the batch axes
188    head_full = reshard(params["head"].astype(compute_dtype), REPLICATED)
189    return rms_norm(h) @ head_full

193    logits = forward(params, x, compute_dtype=compute_dtype).astype(jnp.float32)

208def adam_update(params, opt_state, grads, *, lr, b1=0.9, b2=0.95, eps=1e-8):
209    count = opt_state["count"] + 1
210    bc1 = 1.0 - b1 ** count.astype(jnp.float32)
211    bc2 = 1.0 - b2 ** count.astype(jnp.float32)
212 
213    def upd(p, m, v, g):
214        g = g.astype(jnp.float32)
215        m = b1 * m + (1.0 - b1) * g
216        v = b2 * v + (1.0 - b2) * jnp.square(g)
217        p = p - lr * (m / bc1) / (jnp.sqrt(v / bc2) + eps)
218        return p, m, v
219 
220    out = jax.tree.map(upd, params, opt_state["m"], opt_state["v"], grads, is_leaf=lambda t: isinstance(t, jax.Array))
221    new_params = jax.tree.map(lambda t: t[0], out, is_leaf=lambda t: isinstance(t, tuple))
222    new_m = jax.tree.map(lambda t: t[1], out, is_leaf=lambda t: isinstance(t, tuple))
223    new_v = jax.tree.map(lambda t: t[2], out, is_leaf=lambda t: isinstance(t, tuple))
224    return new_params, {"m": new_m, "v": new_v, "count": count}
