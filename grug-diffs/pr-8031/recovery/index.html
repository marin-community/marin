118class RMSNorm(eqx.Module):
119    weight: jax.Array
120    eps: float = eqx.field(static=True)
121 
122    @staticmethod
123    def init(dim: int, eps: float) -> "RMSNorm":
124        return RMSNorm(weight=jnp.ones((dim,), dtype=jnp.float32), eps=eps)
125 
126    @named_call
127    def __call__(self, x: Float[Array, "... D"]) -> Float[Array, "... D"]:
128        weight = unshard(self.weight)
129        dtype = x.dtype
130        x = x.astype(jnp.float32)
131        variance = jnp.mean(jnp.square(x), axis=-1, keepdims=True)
132        normed = x * jax.lax.rsqrt(variance + self.eps)
133        return (normed * weight).astype(dtype)
