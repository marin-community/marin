16training mask excludes invalid next-token labels. Copy query tiles of 128 bound
17score storage; arithmetic remains quadratic in sequence length. Generation and
18KV-cache decoding are not implemented.

23    query_tile: int,

26

28    (hidden,). Integer tokens, next-token labels and document IDs have shape
29    (batch, sequence). Copy positions include the observed query token itself,
30    exclude future positions, and must belong to the same packed document.
31    """
32    batch, length, dim = queries.shape
33    if query_tile <= 0 or length % query_tile:
34        raise ValueError("query_tile must be positive and divide sequence length")
35    positions = jnp.arange(length)
36    queries = queries.astype(jnp.float32) / dim**0.5
37    keys, sink = keys.astype(jnp.float32), sink.astype(jnp.float32)
38 
39    @jax.checkpoint
40    def chunk(carry, index):
41        start = index * query_tile
42        query = jax.lax.dynamic_slice_in_dim(queries, start, query_tile, axis=1)
43        segment = jax.lax.dynamic_slice_in_dim(segments, start, query_tile, axis=1)
44        target = jax.lax.dynamic_slice_in_dim(labels, start, query_tile, axis=1)
45        scores = jnp.einsum("bqd,bkd->bqk", query, keys)
46        allowed = (positions[None, None, :] <= (start + jnp.arange(query_tile))[None, :, None]) & (
47            segments[:, None, :] == segment[:, :, None]
48        )
49        scores = jnp.where(allowed, scores, -1e30)
50        sink_score = jnp.einsum("bqd,d->bq", query, sink)
51        normalizer = jnp.logaddexp(jax.nn.logsumexp(scores, axis=-1), sink_score)
52        matches = allowed & (tokens[:, None, :] == target[:, :, None])
53        # Finite sentinel avoids undefined gradients of an all-minus-inf
54        # logsumexp when the target never occurs in the visible document.
55        target_score = jax.nn.logsumexp(jnp.where(matches, scores, -1e30), axis=-1)
56        return carry, (sink_score - normalizer, target_score - normalizer)

327def _make_train_step(
328    optimizer: optax.GradientTransformation,
329    mp: jmp.Policy,
330    *,
331    z_loss_weight: float,
332    loss_implementation: str | tuple[str, ...] | None = None,
333    ema_beta: float | None,
334    watch_config: WatchConfig | None = None,
335    backward_flow_config: BackwardFlowConfig | None = None,
336):
337    one = jnp.array(1, dtype=jnp.int32)
338    z_loss = z_loss_weight if z_loss_weight > 0 else None
339    if watch_config is not None:
340        if isinstance(watch_config.watch_targets, str):
341            watch_targets = tuple(t.strip() for t in watch_config.watch_targets.split(","))
342        else:
343            watch_targets = tuple(watch_config.watch_targets)
344    else:
345        watch_targets = ()
346 
347    @functools.partial(jax.jit, donate_argnums=(0,), static_argnames=("compute_backward_flow", "compute_watch"))
348    def train_step(
349        state: GrugTrainState,
350        batch,
351        *,
352        compute_watch: bool = False,
353        compute_backward_flow: bool = False,
354    ):
355        def loss_fn(params):
356            compute_params = mp.cast_to_compute(params)
357            return compute_params.next_token_loss(
358                batch.tokens,
359                batch.loss_weight,
360                mask=batch.attn_mask,
361                reduction="mean",
362                logsumexp_weight=z_loss,
363                loss_implementation=loss_implementation,
364            )
365 
366        backward_flow_stats = None
367        if backward_flow_config is not None and compute_backward_flow:
368            gradient_scale = jnp.sum(jnp.asarray(batch.loss_weight, dtype=jnp.float32))
369            with capture_backward_flow(backward_flow_config, gradient_scale=gradient_scale):
370                with levanter.tracker.defer_tracker_for_jit() as backward_flow_stats:
371                    loss, grads = jax.value_and_grad(loss_fn)(state.params)
372        else:
373            loss, grads = jax.value_and_grad(loss_fn)(state.params)
374        updates, opt_state = optimizer.update(grads, state.opt_state, state.params)
375        params = optax.apply_updates(state.params, updates)
376 
377        if ema_beta is None:
378            ema_params = None
379        else:
380            if state.ema_params is None:
381                raise ValueError("ema_params must be initialized when ema_beta is set.")
382            ema_params = jax.tree_util.tree_map(
383                lambda old, new: ema_beta * old + (1.0 - ema_beta) * new,
384                state.ema_params,
385                params,
386            )
387 
388        watch_stats = None
389        if watch_config is not None and compute_watch:
390            watch_stats = compute_watch_stats(
391                watch_targets=watch_targets,
392                include_norms=watch_config.include_norms,
393                include_per_parameter_norms=watch_config.include_per_parameter_norms,
394                include_histogram=watch_config.include_histograms,
395                split_scan_layers=watch_config.split_scan_layers,
396                params=state.params,
397                grads=grads,
398                updates=updates,
399                opt_state=state.opt_state,
400                model_tree_type=type(state.params),
401            )
402 
403        next_state = dataclasses.replace(
404            state,
405            step=state.step + one,
406            params=params,
407            opt_state=opt_state,
408            ema_params=ema_params,
409        )
410 
411        metrics = {"train/loss": loss}
412        if backward_flow_stats is not None:
413            metrics[_BACKWARD_FLOW_METRICS_KEY] = backward_flow_stats
414 
415        return next_state, metrics, watch_stats
416 
417    return train_step
