236) -> tuple[GrugMoePipelineStage, ...]:
237    """Split a Grug MoE transformer into contiguous stage pytrees."""
238    if layer_counts is None:
239        ranges = evenly_partition_layers(len(model.blocks), num_stages)
240    else:
241        if len(layer_counts) != num_stages:

246            raise ValueError(f"layer counts sum to {sum(layer_counts)}, but model has {len(model.blocks)} layers")
247        boundaries = np.cumsum((0, *layer_counts))

453def microbatched_staged_loss(

455    batch: GrugLmExample,
456    *,
457    num_microbatches: int,
458    logsumexp_weight: float | None = None,
459) -> jax.Array:
460    """Return the loss after sequential microbatch accumulation."""
461    cross_entropy_sum = jnp.array(0.0, dtype=jnp.float32)

414

459        )

513            )

521 
522        state_callbacks = StateCallbackRunner[GrugTrainState](
523            step_getter=lambda s: s.step,
524            model_getter=lambda s: s.params,
525            eval_model_getter=lambda s: s.ema_params if s.ema_params is not None else s.params,
526            opt_state_getter=lambda s: s.opt_state,
527        )
528        state_callbacks.add_hook(
529            callbacks.log_performance_stats(config.model.max_seq_len, batch_schedule, flops_per_example),
530            every=log_every,
531        )
532        state_callbacks.add_hook(callbacks.pbar_logger(total=trainer.num_train_steps), every=log_every)
533        state_callbacks.add_hook(callbacks.log_step_info(trainer.num_train_steps), every=log_every)

572                )

574                    backward_flow_config is not None
575                    and backward_flow_config.interval > 0
576                    and current_step % backward_flow_config.interval == 0
577                )
578                state, metrics, watch_stats = train_step(
579                    state,
580                    batch,
581                    compute_watch=compute_watch,
582                    compute_backward_flow=compute_backward_flow,
583                )
584                backward_flow_stats = metrics.get(_BACKWARD_FLOW_METRICS_KEY)
585                if backward_flow_stats is not None:
586                    metrics = {key: value for key, value in metrics.items() if key != _BACKWARD_FLOW_METRICS_KEY}
587                step = int(state.step) - 1
588 
589                jax.block_until_ready(metrics["train/loss"])

615                            )

619                            backward_flow_metrics=backward_flow_stats,

621                            run_id=run_id,
622                            step=step,
623                        )
624                        levanter.tracker.log(

627                        )

634            )

636        else:

640                checkpointer.on_step(tree=state, step=int(state.step), force=True)
641                checkpointer.wait_until_finished()
642 
643    levanter.tracker.current_tracker().finish()
