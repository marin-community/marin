3Canonical pipeline-parallel Grug MoE implementation. This directory is a copy-paste

8[`pipeline.py`](./pipeline.py) splits the local transformer into stage pytrees and
9builds automatic JaxPP ZeroBubble or DualPipeV optimizer steps. [`train.py`](./train.py)
10owns model initialization, the training loop, and the Fray dispatch entry point.
11[`benchmark.py`](./benchmark.py) maps environment variables onto that trainer for

13`PIPELINE_SCHEDULE=automatic_zero_bubble`. Set `PIPELINE_SCHEDULE=automatic_dualpipe_v`,
14`PIPELINE_STAGES=2P`, and `PIPELINE_PHYSICAL_STAGES=P` to fold two logical stages onto

17stage.

233def split_transformer(
234    model: Transformer,
235    num_stages: int,
236    *,
237    layer_counts: tuple[int, ...] | None = None,
238) -> tuple[GrugMoePipelineStage, ...]:
239    """Split a Grug MoE transformer into contiguous stage pytrees."""
240    if layer_counts is None:
241        ranges = evenly_partition_layers(len(model.blocks), num_stages)
242    else:
243        if len(layer_counts) != num_stages:
244            raise ValueError(f"expected {num_stages} layer counts, got {len(layer_counts)}")
245        if any(count <= 0 for count in layer_counts):
246            raise ValueError(f"layer counts must be positive, got {layer_counts}")
247        if sum(layer_counts) != len(model.blocks):
248            raise ValueError(f"layer counts sum to {sum(layer_counts)}, but model has {len(model.blocks)} layers")
249        boundaries = np.cumsum((0, *layer_counts))
250        ranges = tuple((int(start), int(end)) for start, end in itertools.pairwise(boundaries))
251    stages = []
252    for stage_index, (start_layer, end_layer) in enumerate(ranges):
253        is_first = stage_index == 0
254        is_last = stage_index == num_stages - 1
255        stages.append(
256            GrugMoePipelineStage(
257                token_embed=model.token_embed if is_first else None,
258                embed_norm=model.embed_norm if is_first else None,
259                embed_gated_norm=model.embed_gated_norm if is_first else None,
260                output_proj=model.output_proj if is_last else None,
261                blocks=model.blocks[start_layer:end_layer],
262                final_norm=model.final_norm if is_last else None,
263                final_gated_norm=model.final_gated_norm if is_last else None,
264                config=model.config,
265                start_layer=start_layer,
266                end_layer=end_layer,

268        )
