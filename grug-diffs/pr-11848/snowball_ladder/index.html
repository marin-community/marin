4"""Build fixed-size, pre-mixed token caches for the Snowball scaling ladder.
5 
6The ladder trains on a GPU cluster that cannot stream the full datakit stores, so each data phase
7becomes one flat Levanter cache, built once next to its source store and copied to the cluster.
8Training reads it as a single ``DatasetComponent(flat_cache=True)`` at weight 1.
9 
10Sampling a few percent of documents still touches nearly every 256k-token chunk of the source, so a
11build reads about the whole source store. Run it in the store's region.
12 
13Each recipe samples whole documents from a 200-cell ``cluster=N/quality=Q`` store. Tokens are
14copied as stored, so there is no decode or retokenization and every document keeps its EOS (the
15loader derives document-masking segment ids from it). Documents are not length-capped.
16 
17For cell ``c`` with ``A_c`` stored tokens, output cache ``k`` wants ``p_k = target_k * w_k[c] / A_c``
18copies of each document. The caches take consecutive bands ``[a_k, a_k + p_k)`` on one per-document
19uniform draw ``u``, and a document lands in cache ``k`` ``ceil(b_k - u) - ceil(a_k - u)`` times. So
20the expected tokens per cell hit the mixture exactly, the caches share no documents while the
21bands fit in ``[0, 1)``, and a cell that is too small repeats documents instead of falling short.
22The draw is seeded by (seed, cell, shard), so results do not depend on how work is split.
23 
24Each work item is a run of one cell's shards expected to yield about ``PART_TOKENS`` output tokens,
25and writes one part per cache, which keeps the copied caches to a few thousand parts.

146def resolve_cell_weights(
147    weights: dict[str, float], groups: dict[str, tuple[str, ...]], cell_tokens: dict[str, int]
148) -> dict[str, float]:
149    """Expand groups to member cells and normalize to a simplex over cells."""
150    resolved: dict[str, float] = {}
151    for name, weight in weights.items():
152        if weight <= 0:
153            continue
154        if name in groups:
155            members = groups[name]
156            total = sum(cell_tokens[m] for m in members)
157            for member in members:
158                resolved[member] = resolved.get(member, 0.0) + weight * cell_tokens[member] / total
159        else:
160            if name not in cell_tokens:
161                raise ValueError(f"weight for {name} but the store has no such cell")
162            resolved[name] = resolved.get(name, 0.0) + weight
163    total = sum(resolved.values())
164    return {cell: w / total for cell, w in resolved.items()}

167def cell_bands(

220def group_shards(shards: Sequence[dict], copies_per_token: float, part_tokens: int) -> list[list[dict]]:

286                "expected_tokens": {name: (b - a) * cell_tokens[cell] for name, (a, b) in bands.get(cell, {}).items()},
287            }
288            for cell in cells

341def _read_docs(store: JaggedArrayStore, indices: np.ndarray, lengths: np.ndarray) -> Iterator[np.ndarray]:
342    """Yield documents in ``indices`` order, batching reads by document count and tokens.
343 
344    ``lengths[i]`` is the length of document ``indices[i]``.
345    """
346    start = 0
347    while start < len(indices):
348        stop = start
349        batch_tokens = 0
350        while (
351            stop < len(indices)
352            and stop - start < READ_BATCH_DOCS
353            and (stop == start or batch_tokens < READ_BATCH_TOKENS)
354        ):
355            batch_tokens += int(lengths[stop])
356            stop += 1
357        for doc in store.get_batch_sync(indices[start:stop].tolist()):
358            yield np.asarray(doc, dtype=np.int32)
359        start = stop

505def validate(out: str, measure_bytes: bool, sample_docs: int = 2_000) -> None:
506    the_plan = load_plan(out)

512        if len(loaded) != stats["total_elements"]:
513            raise ValueError(f"{cache}: ledger has {len(loaded)} docs, report has {stats['total_elements']}")
514        if stats["realized_over_target"] < 1.0:
515            logger.warning("%s: realized tokens are %.4f of target", cache, stats["realized_over_target"])
516        rng = np.random.default_rng(0)
517        picks = np.sort(rng.choice(len(loaded), size=min(sample_docs, len(loaded)), replace=False))
518        docs = [np.asarray(d["input_ids"]) for d in loaded.get_batch_sync(picks.tolist())]
519        last_tokens, counts = np.unique([int(d[-1]) for d in docs], return_counts=True)
520        eos_share = counts.max() / len(docs)
521        logger.info(
522            "%s: %d sampled docs, max token id %d, most common final token %d on %.1f%% of docs",
523            cache,
524            len(docs),
525            max(int(d.max()) for d in docs),
526            int(last_tokens[counts.argmax()]),
527            100 * eos_share,
528        )
529        if min(len(d) for d in docs) == 0 or eos_share < 0.99:
530            raise ValueError(f"{cache}: empty documents or documents without a shared EOS")
531        if measure_bytes:
532            size = _cache_bytes(root)
533            stats["bytes"] = size
534            logger.info("%s: %.2f TB on disk, %.2f bytes per token", cache, size / 1e12, size / stats["total_tokens"])
535    deviations = [
536        (abs(v["realized"] / v["expected"] - 1), cell, cache)
537        for cell, caches in report["cells"].items()
538        for cache, v in caches.items()
539        if v["expected"] >= 1e9
540    ]
541    if deviations:
542        worst = max(deviations)
543        logger.info(
544            "largest per-cell deviation among cells with >=1B expected tokens: %.2f%% (%s, %s)",
545            100 * worst[0],
546            worst[1],
547            worst[2],
548        )
549    if measure_bytes:
550        StoragePath(prefix_join(out, REPORT_FILE)).write_text(json.dumps(report, indent=1))
551    logger.info("VALIDATE OK")
