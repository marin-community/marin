15then to the coda). Let `e` be the prelude's output state. The core receives
16`h₀ = 0`, so its entry state is `BO(0, e) = α·e` (one RMSNorm is inert on the

20byte-identical to the parent template (regression-tested against its logits).

95def _manual_boundary_forward(
96    model: Transformer,
97    tokens: jax.Array,
98    *,
99    e_source,
100    core_entry,
101    coda_entry,
102) -> jax.Array:
103    """Execute the paper's prelude / core / coda pipeline by hand over the model's own submodules.
104 
105    Weights are shared with the model under test, so what gets checked is the wiring of
106    arXiv 2609.19107 Eq. 2/3 (K = 1): ``e_source(embed, prelude_out)`` picks what ``e``
107    is, ``core_entry(e, prelude_out)`` computes the state entering the core, and
108    ``coda_entry(e, core_out)`` the state entering the coda.

111    embed = model.embed_gated_norm(model.embed_norm(unshard(model.token_embed)[tokens]))
112    num_blocks = len(model.blocks)
113    prelude_len, coda_len = cfg.prelude_len, cfg.coda_len
114    core_len = num_blocks - prelude_len - coda_len
115 
116    hidden = embed
117    for i in range(prelude_len):
118        hidden = _run_block(model, i, hidden)
119    prelude_out = hidden
120 
121    e = e_source(embed, prelude_out)
122    hidden = core_entry(e, prelude_out)
123    for i in range(prelude_len, prelude_len + core_len):
124        hidden = _run_block(model, i, hidden)
125    core_out = hidden
126 
127    hidden = coda_entry(e, core_out)
128    for i in range(prelude_len + core_len, num_blocks):
129        hidden = _run_block(model, i, hidden)
130    return model.final_gated_norm(model.final_norm(hidden))
