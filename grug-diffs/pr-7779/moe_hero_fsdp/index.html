53### Effective speedup calculation
54 
55The question: how much faster (in wall-clock time) does the variant reach
56its final loss compared to how long the baseline would need to reach that
57same loss?
58 
59The scaling law `loss(C) = 1.6 + A · C^(-0.0941)` has a fixed exponent but
60the coefficient A varies. We recenter A to pass through the baseline's
61actual (budget, loss) point, then invert to find how much compute the
62baseline would need to match the variant's loss. Finally, compare wall-clock
63times.
64 
65```python
66def effective_speedup(baseline_loss, baseline_tps, variant_loss, variant_tps, budget):
67    """Wall-clock speedup of the variant over the baseline.
68 
69    Returns > 1 if the variant reaches its loss faster in real time.
70    """
71    L_inf = 1.6
72    alpha = 0.0941
73 
74    # 1. Fit the scaling law coefficient through the baseline's data point.
75    #    loss = L_inf + A_bl * C^(-alpha)  =>  A_bl = (loss - L_inf) * C^alpha
76    A_bl = (baseline_loss - L_inf) * budget ** alpha
77 
78    # 2. How much compute would the baseline need to reach variant's loss?
79    C_needed = (A_bl / (variant_loss - L_inf)) ** (1 / alpha)
80 
81    # 3. Compare wall-clock times:
82    #    baseline_time = C_needed / baseline_tps
83    #    variant_time  = budget / variant_tps
84    return (C_needed / baseline_tps) / (budget / variant_tps)
85```

66def compute_tokens_and_batch(
67    budget: float,
68    flops_per_token: float,
69    target_steps: int = DEFAULT_TARGET_STEPS,
70    min_batch_size: int = MIN_BATCH_SIZE,
71    seq_len: int = SEQ_LEN,
72) -> tuple[float, int, int]:
73    """Derive (tokens, batch_size, num_steps) from a compute budget and FLOPs-per-token.
74 
75    ``seq_len`` controls the sequence length used to convert between batch_size
76    (sequences) and tokens_per_batch (tokens per step). All downstream formulas
77    that depend on batch size use ``tokens_per_batch = batch_size * seq_len``
78    so they work correctly at any sequence length.
79    """
80    tokens = budget / (3 * flops_per_token)
81    batch_exact = tokens / (target_steps * seq_len)
82    batch_size = max(min_batch_size, _round_to_power_of_two(batch_exact))
83    train_steps = max(1, round(tokens / (batch_size * seq_len)))
84    return tokens, batch_size, train_steps

87@dataclass(frozen=True)
88class MoeHeuristic:
89    """Compute-scaling heuristic for MoE models (May Recipe refit).
90 
91    Returns a ``MuonH`` optimizer config via ``build_optimizer_config``.
92 
93    ``adam_lr  = lr_coeff * tokens^lr_tokens_exp * dim^lr_dim_exp * sqrt(tokens_per_batch)``
94    ``muonh_lr = muonh_ratio * adam_lr``
95    ``C       = 3 * flops_per_token * tokens``  (flops_per_token excludes lm_head)
96    """
97 
98    # --- LR scaling ---
99    # May Recipe refit (issue #5951; 17 cells across d{512,768,1024,1280},
100    # R^2=0.996):  muonh_lr = 18.31 * tokens^-0.395 * dim^-0.150 * sqrt(B).
101    lr_coeff: float = 0.06602
102    lr_tokens_exp: float = -0.395
103    lr_dim_exp: float = -0.150
104    muonh_ratio: float = 13 / 3
105 
106    # --- Base hyperparameters ---
107    epsilon_coeff: float = 9.676e-18
108    beta1: float = 0.9062
109    beta2_base: float = 0.999
110    beta2_reference_tpb: int = 131_072  # beta2 = beta2_base^(tpb / beta2_reference_tpb)
111 
112    # --- Fixed hyperparameters ---
113    max_grad_norm: float = 1.0
114 
115    # --- Schedule ---
116    min_lr_ratio: float = 0.05
117    lr_schedule: str = "linear"
118    decay: float | None = None
119 
120    # --- Architecture ---
121    vocab_size: int = 128_256
122    hidden_head_ratio: int = 128
123    gqa_ratio: int | None = 4  # None = MHA, 4 = 4:1 GQA, etc.
124    base_hidden_layer_ratio: int = 64
125    layer_scaling_factor: float = 4.0
126    layer_formula_offset: int = 9
127 
128    # --- Constraints ---
129    max_learning_rate: float = 0.05
130    min_beta2: float = 0.95
131    max_beta2: float = 0.9999
132 
133    def _compute_adam_lr(self, tokens_per_batch: int, tokens: float, hidden_dim: int) -> float:
134        """adam_lr = lr_coeff * tokens^lr_tokens_exp * dim^lr_dim_exp * sqrt(tokens_per_batch)"""
135        adam_lr = (
136            self.lr_coeff * (tokens**self.lr_tokens_exp) * (hidden_dim**self.lr_dim_exp) * math.sqrt(tokens_per_batch)
137        )
138        return min(self.max_learning_rate, adam_lr)
139 
140    def _compute_learning_rate(self, tokens_per_batch: int, tokens: float, hidden_dim: int) -> float:
141        """muonh_lr = muonh_ratio * adam_lr"""
142        adam_lr = self._compute_adam_lr(tokens_per_batch, tokens, hidden_dim)
143        return min(self.max_learning_rate, self.muonh_ratio * adam_lr)
144 
145    def _compute_epsilon(self, tokens_per_batch: int, tokens: float) -> float:
146        """epsilon = epsilon_coeff * sqrt(tokens / tokens_per_batch)"""
147        return self.epsilon_coeff * math.sqrt(tokens / tokens_per_batch)
148 
149    def _compute_beta2(self, tokens_per_batch: int) -> float:
150        """beta2 = clip(beta2_0^(tpb/tpb0), min_beta2, max_beta2). Constant token half-life."""
151        exponent = tokens_per_batch / self.beta2_reference_tpb
152        return max(self.min_beta2, min(self.max_beta2, self.beta2_base**exponent))

154    def build_optimizer_config(

161        scaling is the May Recipe refit (see module docstring).
