 assert fs.cat_file(path)==data
 if local:(p/name).write_bytes(data)

def evaluate(i,row):
 payload={'model':'glm-5.3','messages':row['messages'],'chat_template_kwargs':{'reasoning_effort':'medium'},'max_tokens':16384,'thinking_token_budget':8192,'temperature':0,'top_p':1,'seed':42}
 save(f'request-{i}.json',payload)
 request=urllib.request.Request(os.environ['LABBENCH_GLM_BASE']+'/chat/completions',data=json.dumps(payload).encode(),headers={'Authorization':'Bearer '+os.environ['OPENAI_API_KEY'],'Content-Type':'application/json'})
 begin=time.monotonic()
 http_status=None
 try:
  try:
   response=urllib.request.urlopen(request,timeout=1200)
  except urllib.error.HTTPError as exc:
   response=exc
  with response:
   http_status=response.status
   headers={k:v for k,v in response.headers.items() if k.lower() in {'content-type','content-length','content-encoding','transfer-encoding','date','server','cf-ray','x-request-id','request-id','x-openrouter-request-id','x-generation-id'}}
   save(f'http-{i}.json',{'status':http_status,'headers':headers})
   try:
    body=response.read()
   except http.client.IncompleteRead as exc:
    save(f'body-{i}.bin',exc.partial)
    raise
   save(f'body-{i}.bin',body)
   if isinstance(response,urllib.error.HTTPError):raise response
  raw=json.loads(body)
 except (urllib.error.URLError,TimeoutError,http.client.HTTPException,json.JSONDecodeError,UnicodeDecodeError) as exc:
  failure={'index':i,'task':row['task'],'subtask':row['subtask'],'doc_id':row['doc_id'],'status':'request_failed','exception':type(exc).__name__,'http_status':http_status,'elapsed_seconds':time.monotonic()-begin}
  if isinstance(exc,json.JSONDecodeError):
   failure.update(json_error=exc.msg,error_position=exc.pos,error_line=exc.lineno,error_column=exc.colno)
  if isinstance(exc,http.client.IncompleteRead):
   failure.update(partial_body=True,captured_bytes=len(exc.partial),remaining_expected_bytes=exc.expected)
  save(f'failure-{i}.json',failure)
  return failure
 elapsed=time.monotonic()-begin
 save(f'response-{i}.json',{'input':row,'response':raw,'elapsed_seconds':elapsed,'request_sha256':hashlib.sha256(json.dumps(payload,sort_keys=True).encode()).hexdigest()})
 if raw.get('error') or not raw.get('choices'):
  failure={'index':i,'task':row['task'],'subtask':row['subtask'],'doc_id':row['doc_id'],'status':'provider_error','elapsed_seconds':elapsed}
  save(f'failure-{i}.json',failure);return failure
 completion=completion_response_from_chat_choice(raw,raw['choices'][0])
 response=CompletionText(completion.normalized_content(),completion,CompletionContentPolicy.COMBINE)
 spec=EvalSpec(task=row['task'],source=SourceRef(DATASET,'train',REVISION),source_id=row['doc_id'],ordinal=i,requests=(),grader=GraderSpec('letter_choice',reference={'extractor':'hle'}),target=row['target'])
 grade=letter_choice(spec,[response])
 scored={'index':i,'task':row['task'],'subtask':row['subtask'],'doc_id':row['doc_id'],'status':'scored','prompt_sha256':row['prompt_sha256'],'response_sha256':hashlib.sha256(json.dumps(raw,sort_keys=True).encode()).hexdigest(),'target':row['target'],'extracted':grade.filtered_resps[0],'accuracy':grade.metrics['accuracy'],'classification':str(completion.classification),'finish_reason':completion.finish_reason,'elapsed_seconds':elapsed,'usage':raw.get('usage'),'model':raw.get('model'),'provider':raw.get('provider')}
 save(f'grade-{i}.json',scored)
 return scored
