def _xsa_ref(y, v):
    dot = jnp.sum(y * v, axis=-1, keepdims=True)
    nsq = jnp.sum(v * v, axis=-1, keepdims=True)
    return y - (dot / (nsq + 1e-6)) * v

@jax.custom_vjp
def xsa_bwd32(y, v):
    return _xsa_ref(y, v)
def _x_fwd(y, v):
    return _xsa_ref(y, v), (y, v)
def _x_bwd(res, g):
    y, v = res
    _, vjp = jax.vjp(_xsa_ref, y.astype(jnp.float32), v.astype(jnp.float32))
    gy, gv = vjp(g.astype(jnp.float32))
    return gy.astype(y.dtype), gv.astype(v.dtype)
xsa_bwd32.defvjp(_x_fwd, _x_bwd)

def _rms_eps(x, eps):
    variance = jnp.mean(jnp.square(x.astype(jnp.float32)), axis=-1, keepdims=True)
    return (x * jax.lax.rsqrt(variance + eps)).astype(x.dtype)

@jax.custom_vjp
def rms_bwd_reg(x):
    return _rms_eps(x, 1e-6)
def _r_fwd(x):
    return _rms_eps(x, 1e-6), x
def _r_bwd(x, g):
    _, vjp = jax.vjp(lambda a: _rms_eps(a, 1e-2), x)
    return vjp(g)
rms_bwd_reg.defvjp(_r_fwd, _r_bwd)
