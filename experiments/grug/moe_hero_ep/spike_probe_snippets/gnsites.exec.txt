# Which GatedNorm carries the amplification: stop the gate's gradient at one site at a time (forward unchanged).
import inspect, re, textwrap
from experiments.grug.moe_hero_ep import model as M
orig = {"block": M.Block.__call__, "tf": M.Transformer.__call__}

def gn_sg(gn, x):
    gate_hidden = jax.nn.silu(jnp.einsum("...d,dr->...r", x, gn.w_down))
    gate = jax.nn.sigmoid(jnp.einsum("...r,rd->...d", gate_hidden, gn.w_up))
    return x * jax.lax.stop_gradient(gate.astype(x.dtype))

def patched(fn, old, new):
    src = textwrap.dedent(inspect.getsource(fn)); assert old in src, old
    ns = dict(M.__dict__); ns["_GN_SG"] = gn_sg; exec(src.replace(old, new), ns); return ns["__call__"]

A = "attn_in = self.attn_gated_norm(self.rms_attn(x))"

F = "hidden = self.final_gated_norm(self.final_norm(hidden))"

    "embed_gn_sg": {"tf": patched(orig["tf"], E, "hidden = _GN_SG(self.embed_gated_norm, self.embed_norm(hidden))")},
    "final_gn_sg": {"tf": patched(orig["tf"], F, "hidden = _GN_SG(self.final_gated_norm, self.final_norm(hidden))")},
}
try:
