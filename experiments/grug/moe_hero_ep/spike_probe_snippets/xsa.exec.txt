# XSA numerics test: patch CausalSelfAttention.__call__'s XSA projection (eps / fp32) and recompute the
# sequence's gradient with the hero's SM100 kernels. Also record value-norm statistics. Inputs: STEP, RACK, SEQ.
import inspect, re, textwrap
from experiments.grug.moe_hero_ep import model as M
_orig_call = M.CausalSelfAttention.__call__
_src = textwrap.dedent(inspect.getsource(_orig_call))
OLD_DOT = "dot = jnp.sum(attn_out * aligned_v, axis=-1, keepdims=True)"
OLD_NSQ = "v_norm_sq = jnp.sum(aligned_v * aligned_v, axis=-1, keepdims=True)"
OLD_PROJ = "attn_out = attn_out - (dot / (v_norm_sq + 1e-6)) * aligned_v"
assert OLD_DOT in _src and OLD_NSQ in _src and OLD_PROJ in _src
vstats = []

def _vrec(vmin, small, amp):
    vstats.append((np.asarray(vmin).tolist(), np.asarray(small).tolist(), np.asarray(amp).tolist()))

variants = {
    "baseline_eps1e-6_bf16": _src,
    "fp32_eps1e-6": _src.replace(OLD_DOT, "dot = jnp.sum(attn_out.astype(jnp.float32) * aligned_v.astype(jnp.float32), axis=-1, keepdims=True)")
        .replace(OLD_NSQ, "v_norm_sq = jnp.sum(jnp.square(aligned_v.astype(jnp.float32)), axis=-1, keepdims=True)")
        .replace(OLD_PROJ, "attn_out = (attn_out.astype(jnp.float32) - (dot / (v_norm_sq + 1e-6)) * aligned_v.astype(jnp.float32)).astype(attn_out.dtype)"),
    "bf16_eps1e-2": _src.replace("(v_norm_sq + 1e-6)", "(v_norm_sq + 1e-2)"),
    "record_vnorms": _src.replace(OLD_PROJ, OLD_PROJ + "\n    _vf = v_norm_sq.astype(jnp.float32)[..., 0]\n    _amp = jnp.sqrt(_vf) / (_vf + 1e-6)\n    jax.debug.callback(_XSA_REC, jnp.min(_vf, axis=(1, 2)), jnp.sum((_vf < 1e-4).astype(jnp.int32), axis=(1, 2)), jnp.max(_amp, axis=(1, 2)))"),
}
try:
    for name, src in variants.items():
        ns = dict(M.__dict__); ns["_XSA_REC"] = _vrec
        try:
            exec(src, ns)
        except Exception as exc:
            result[name] = {"error": repr(exc)[:500]}
            continue
        M.CausalSelfAttention.__call__ = ns["__call__"]
        jax.clear_caches()
        vstats.clear()
        loss, denom, _, g = probe.slice_grad(STEP, RACK, [SEQ])
        n = probe.norms(g, 1.0); del g
        layers = [sum(v * v for k, v in n.items() if re.match(rf"grad/norm/stacked_blocks\.{l}\.", k)) ** 0.5 for l in range(48)]
        result[name] = {"loss": loss, "grad_total": n["grad/norm/total"], "layers": layers}
        if name == "record_vnorms":
            pid = jax.process_index()
            from rigging.filesystem.storage_path import StoragePath
            import json as _json
            StoragePath(f"{runner.output}/results/vnorm-p{pid:02d}.json").write_text(_json.dumps(vstats[:48]))
finally:
    M.CausalSelfAttention.__call__ = _orig_call
    jax.clear_caches()
