# Backward taps: per-token cotangent norms for row SEQ at every op boundary of every layer, plus the top tokens'
# cotangent and forward vectors for layers in VEC_LAYERS. Forward math is unchanged (identity custom_vjp taps).
# Inputs: STEP, RACK, SEQ, TAG, VEC_LAYERS, TOPK.
import functools, inspect, io, textwrap

B = 256
onehot = jax.nn.one_hot(SEQ, B, dtype=jnp.float32)
LAYER = [None]
REC = []
IS0 = jax.process_index() == 0

def pick_row(x):
    x3 = x.reshape(x.shape[0], x.shape[1], -1)
    return jnp.einsum("b,bsf->sf", onehot, x3.astype(jnp.float32), out_sharding=P())

def _rec(name, idx, norms, top, gvec, xvec):
    if not IS0:
        return
    idx = int(idx)
    entry = {"name": name, "layer": idx, "norms": np.asarray(norms, np.float32), "top": np.asarray(top)}
    if idx in VEC_LAYERS:
        entry["gvec"] = np.asarray(gvec, np.float32); entry["xvec"] = np.asarray(xvec, np.float32)
    REC.append(entry)

TAPS = {}
def make_tap(name):
    @jax.custom_vjp
    def tap(x, idx):
        return x
    def fwd(x, idx):
        return x, (pick_row(x), idx)
    def bwd(res, g):
        xr, idx = res
        gr = pick_row(g)
        n = jnp.sqrt(jnp.sum(gr * gr, axis=-1))
        _, top = jax.lax.top_k(n, TOPK)
        jax.debug.callback(functools.partial(_rec, name), idx, n, top, gr[top], xr[top])
        return g, None
    tap.defvjp(fwd, bwd)
    return tap

def T(x, name):
    if name not in TAPS:
        TAPS[name] = make_tap(name)
    return TAPS[name](x, LAYER[0])

def gn_tapped(gn, x):
    gate_hidden = jax.nn.silu(jnp.einsum("...d,dr->...r", x, gn.w_down))
    pre = T(jnp.einsum("...r,rd->...d", gate_hidden, gn.w_up), "gn_preact")
    gate = jax.nn.sigmoid(pre)
    return T(x, "gn_value") * gate.astype(x.dtype)

def patched(fn, reps, extra):
    src = textwrap.dedent(inspect.getsource(fn))
    for a, b in reps:
        assert a in src, a
        src = src.replace(a, b)
    ns = dict(M.__dict__); ns.update(extra); exec(src, ns); return ns["__call__"]

     'return _T(jnp.einsum("bsh,hd->bsd", attn_out, self.w_o, out_sharding=P(_BATCH_AXES, residual_seq_axis, None)), "attn_out")'),
